cmake_minimum_required(VERSION 3.16)
project(ControlPanel LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)

find_package(Qt6 REQUIRED COMPONENTS Widgets Multimedia Concurrent DBus OpenGL QuickWidgets)
find_package(KF6WidgetsAddons REQUIRED)
find_package(KF6WindowSystem REQUIRED)
find_package(KF6KCMUtils REQUIRED)
find_package(KF6CoreAddons REQUIRED)
find_package(ZLIB REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(AeroQt REQUIRED)
find_package(Cups REQUIRED)

add_compile_options(-g)

# Generate umbrella header for pages
  file(GLOB ALL_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/ui/pages/*.h")
  set(UMBRELLA_HEADER_CONTENT "// Auto-generated by CMake. Do not edit.\n\n#pragma once\n\n")
  foreach(HEADER_PATH ${ALL_HEADERS})
      get_filename_component(HEADER_NAME "${HEADER_PATH}" NAME)
      if(NOT HEADER_NAME STREQUAL "all.h")
          string(APPEND UMBRELLA_HEADER_CONTENT "#include \"${HEADER_NAME}\"\n")
      endif()
  endforeach()
  string(APPEND UMBRELLA_HEADER_CONTENT "#define FOREACH_PAGE_CLASS(X) \\\n")
  foreach(HEADER_PATH ${ALL_HEADERS})
        get_filename_component(HEADER_CLASS_NAME "${HEADER_PATH}" NAME_WE)
        string(APPEND UMBRELLA_HEADER_CONTENT "\tX(${HEADER_CLASS_NAME}) \\\n")
  endforeach()
  string(APPEND UMBRELLA_HEADER_CONTENT ";\n")
  set(GENERATED_HEADER "${CMAKE_CURRENT_BINARY_DIR}/include/all_pages.h")
  file(WRITE "${GENERATED_HEADER}" "${UMBRELLA_HEADER_CONTENT}")

file(GLOB_RECURSE sources
  "${CMAKE_CURRENT_LIST_DIR}/src/*.cpp"
  "${CMAKE_CURRENT_LIST_DIR}/src/*.h"
  "${CMAKE_CURRENT_LIST_DIR}/src/*.ui"
)

qt_add_executable(controlpanel
  ${sources}
)

target_include_directories(controlpanel PRIVATE
    src/ui
    src/ui/pages
    src/ui/dialogs
    src/devices
    src/data
    src
    "${CMAKE_CURRENT_BINARY_DIR}/include"
)

target_link_libraries(controlpanel PRIVATE
    Qt6::Widgets
    Qt6::Multimedia
    Qt6::Concurrent
    Qt6::DBus
    Qt6::OpenGL
    Qt6::QuickWidgets
    KF6::WidgetsAddons
    KF6::WindowSystem
    KF6::KCMUtils
    KF6::KCMUtilsCore
    KF6::CoreAddons
    ZLIB::ZLIB
    OpenSSL::Crypto
    AeroQt
    Cups::Cups
)

set_target_properties(controlpanel PROPERTIES
    OUTPUT_NAME control
    BUILD_RPATH "$ORIGIN"
    INSTALL_RPATH "$ORIGIN"
)

install(TARGETS controlpanel RUNTIME DESTINATION bin)
install(FILES linux-controlpanel.desktop DESTINATION share/applications)
